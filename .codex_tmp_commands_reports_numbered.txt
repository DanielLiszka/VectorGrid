 109:fn build_report_data_from_dataset_meta(dataset_meta: Option<&Value>) -> Value {
 110:    let Some(dataset_meta) = dataset_meta.and_then(Value::as_object) else {
 111:        return serde_json::json!({
 112:            "meta": Value::Null,
 113:            "ctx": serde_json::Map::<String, Value>::new(),
 114:        });
 115:    };
 116:
 117:    let mut meta = serde_json::Map::<String, Value>::new();
 118:    let mut ctx = serde_json::Map::<String, Value>::new();
 119:
 120:    for key in [
 121:        "timeframe",
 122:        "base_timeframe",
 123:        "derived_timeframe",
 124:        "source",
 125:        "market",
 126:        "symbol",
 127:        "label",
 128:        "csv_path",
 129:        "resample_rule",
 130:        "resample_anchor",
 131:        "source_content_fingerprint",
 132:    ] {
 133:        let Some(value) = dataset_meta.get(key).cloned() else {
 134:            continue;
 135:        };
 136:        let text = value.as_str().map(|text| text.trim()).unwrap_or("");
 137:        if text.is_empty() {
 138:            continue;
 139:        }
 140:        if matches!(key, "label" | "symbol" | "source" | "market" | "csv_path") {
 141:            ctx.insert(key.to_string(), Value::String(text.to_string()));
 142:        } else {
 143:            meta.insert(key.to_string(), Value::String(text.to_string()));
 144:            if matches!(
 145:                key,
 146:                "timeframe"
 147:                    | "base_timeframe"
 148:                    | "derived_timeframe"
 149:                    | "resample_rule"
 150:                    | "resample_anchor"
 151:            ) {
 152:                ctx.insert(key.to_string(), Value::String(text.to_string()));
 153:            }
 154:        }
 155:    }
 156:
 157:    for key in [
 158:        "len",
 159:        "min_ts",
 160:        "max_ts",
 161:        "source_size_bytes",
 162:        "source_mtime_utc_ms",
 163:    ] {
 164:        let Some(value) = dataset_meta.get(key) else {
 165:            continue;
 166:        };
 167:        let Some(number) = value.as_f64().filter(|number| number.is_finite()) else {
 168:            continue;
 169:        };
 170:        if number.fract() == 0.0 {
 171:            meta.insert(
 172:                key.to_string(),
 173:                Value::Number(serde_json::Number::from(number as i64)),
 174:            );
 175:        } else if let Some(number) = serde_json::Number::from_f64(number) {
 176:            meta.insert(key.to_string(), Value::Number(number));
 177:        }
 178:    }
 179:
 180:    for key in ["resample_applied", "adjusted"] {
 181:        if let Some(boolean) = dataset_meta.get(key).and_then(Value::as_bool) {
 182:            ctx.insert(key.to_string(), Value::Bool(boolean));
 183:        }
 184:    }
 185:
 186:    serde_json::json!({
 187:        "meta": if meta.is_empty() { Value::Null } else { Value::Object(meta) },
 188:        "ctx": Value::Object(ctx),
 189:    })
 190:}

 622:    }
