 445:function extractTopRowsFromRunResult(runRes: unknown): Array<{ params?: unknown; metrics?: unknown; mo?: unknown; source_kind?: unknown }> {
 446:  const row = asRecord(runRes)
 447:  if (!row) return []
 448:  const rows: Array<{ params?: unknown; metrics?: unknown; mo?: unknown; source_kind?: unknown }> = []
 449:
 450:  const paretoRowsRaw = Array.isArray(row.pareto) ? row.pareto : []
 451:  if (paretoRowsRaw.length > 0) {
 452:    const pts = paretoRowsRaw
 453:      .map((entry) => {
 454:        const point = asRecord(entry)
 455:        return {
 456:          params: asRecord(point?.params),
 457:          metrics: asRecord(point?.metrics),
 458:          maximize_value: Number(point?.maximize_value ?? Number.NEGATIVE_INFINITY),
 459:          minimize_value: Number(point?.minimize_value ?? Number.POSITIVE_INFINITY),
 460:          violation: Number(point?.violation ?? 0),
 461:        }
 462:      })
 463:      .filter((entry) => !!entry.params)
 464:    pts.sort((left, right) => {
 465:      const leftFeasible = left.violation <= 0
 466:      const rightFeasible = right.violation <= 0
 467:      if (leftFeasible !== rightFeasible) return leftFeasible ? -1 : 1
 468:      if (left.maximize_value !== right.maximize_value) return right.maximize_value - left.maximize_value
 469:      if (left.minimize_value !== right.minimize_value) return left.minimize_value - right.minimize_value
 470:      return 0
 471:    })
 472:    for (const entry of pts) {
 473:      rows.push({
 474:        params: entry.params,
 475:        metrics: entry.metrics,
 476:        mo: {
 477:          maximize_value: entry.maximize_value,
 478:          minimize_value: entry.minimize_value,
 479:          violation: entry.violation,
 480:        },
 481:        source_kind: 'pareto_frontier',
 482:      })
 483:    }
 484:    return rows
 485:  }
 486:
 487:  const topRowsRaw = Array.isArray(row.top) ? row.top : []
 488:  for (const entry of topRowsRaw) {
 489:    if (Array.isArray(entry)) {
 490:      rows.push({
 491:        params: entry[0],
 492:        metrics: entry[1],
 493:        source_kind: 'top_rows',
 494:      })
 495:      continue
 496:    }
 497:    const item = asRecord(entry)

 499:    rows.push({
 500:      params: item.params,
 501:      metrics: item.metrics,
 502:      source_kind: 'top_rows',
 503:    })
 504:  }

 507:  const bestParams = row.best_params
 508:  if (!bestParams || typeof bestParams !== 'object') return []
