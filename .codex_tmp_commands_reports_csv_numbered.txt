 103:    fn parse_csv_fields(line: &str) -> Vec<String> {
 104:        let mut out = Vec::new();
 105:        let mut current = String::new();
 106:        let mut chars = line.chars().peekable();
 107:        let mut in_quotes = false;
 108:
 109:        while let Some(ch) = chars.next() {
 110:            match ch {
 111:                '"' if in_quotes => {
 112:                    if matches!(chars.peek(), Some('"')) {
 113:                        current.push('"');
 114:                        let _ = chars.next();
 115:                    } else {
 116:                        in_quotes = false;
 117:                    }
 118:                }
 119:                '"' => in_quotes = true,
 120:                ',' if !in_quotes => {
 121:                    out.push(current);
 122:                    current = String::new();
 123:                }
 124:                _ => current.push(ch),
 125:            }
 126:        }
 127:
 128:        out.push(current);
 129:        out
 130:    }
