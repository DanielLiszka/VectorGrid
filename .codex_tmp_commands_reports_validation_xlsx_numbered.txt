 214:fn median_option_f64(values: &[Option<f64>]) -> Option<f64> {
 215:    let mut xs: Vec<f64> = values
 216:        .iter()
 217:        .flatten()
 218:        .copied()
 219:        .filter(|value| value.is_finite())
 220:        .collect();
 221:    if xs.is_empty() {
 222:        return None;
 223:    }
 224:    xs.sort_by(|a, b| a.total_cmp(b));
 225:    let mid = xs.len() / 2;
 226:    if xs.len() % 2 == 0 {
 227:        Some((xs[mid - 1] + xs[mid]) / 2.0)
 228:    } else {
 229:        Some(xs[mid])
 230:    }
 231:}

 233:fn approval_label_counts(model: &ValidationExportModel) -> Vec<(String, usize)> {
 234:    use std::collections::BTreeMap;
 235:
 236:    let mut counts: BTreeMap<String, usize> = BTreeMap::new();
 237:    for row in &model.all_rows {
 238:        *counts.entry(row.approval_label.clone()).or_insert(0) += 1;
 239:    }
 240:    if counts.is_empty() {
 241:        if let Some(verdict) = model.run_verdict.as_ref() {
 242:            return vec![(
 243:                format!(
 244:                    "Run Verdict: {}",
 245:                    display_recommendation_label(&verdict.final_recommendation)
 246:                ),
 247:                1,
 248:            )];
 249:        }
 250:    }
 251:
 252:    let mut out = Vec::new();
 253:    for label in [
 254:        "approved",
 255:        "review",
 256:        "rejected",
 257:        "not_approved",
 258:        "compat_shortlist",
 259:        "compat_context",
 260:    ] {
 261:        if let Some(count) = counts.remove(label) {
 262:            out.push((display_approval_label(label).to_string(), count));
 263:        }
 264:    }
 265:    for (label, count) in counts {
 266:        out.push((label, count));
 267:    }
 268:    out
 269:}

 271:fn top_reason_counts(model: &ValidationExportModel, limit: usize) -> Vec<(String, usize)> {
 272:    use std::collections::BTreeMap;
 273:
 274:    let mut counts: BTreeMap<String, usize> = BTreeMap::new();
 275:    for row in &model.all_rows {
 276:        if row.final_approved
 277:            || row.approval_reason.trim().is_empty()
 278:            || row.approval_reason == "approved"
 279:        {
 280:            continue;
 281:        }
 282:        *counts.entry(row.approval_reason.clone()).or_insert(0) += 1;
 283:    }
 284:    let mut out: Vec<(String, usize)> = counts.into_iter().collect();
 285:    out.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
 286:    out.truncate(limit);
 287:    if out.is_empty() {
 288:        if let Some(verdict) = model
 289:            .run_verdict
 290:            .as_ref()
 291:            .filter(|verdict| !verdict.approved)
 292:        {
 293:            out.push((run_level_rejection_reason(verdict), 1));
 294:        }
 295:    }
 296:    out
 297:}

 416:fn write_candidate_rows(
 417:    sheet: &mut Worksheet,
 418:    rows: &[ExportCandidateRow],
 419:    text_fmt: &Format,
 420:    mono_fmt: &Format,
 421:    num_fmt: &Format,
 422:    pct_fmt: &Format,
 423:    int_fmt: &Format,
 424:    approved_fmt: &Format,
 425:    review_fmt: &Format,
 426:    rejected_fmt: &Format,
 427:    compat_fmt: &Format,
 428:) -> Result<(), String> {
 429:    for (idx, row) in rows.iter().enumerate() {
 430:        let r = idx as u32 + 1;
 431:        sheet
 432:            .write_number_with_format(r, COL_RANK, row.rank as f64, int_fmt)
 433:            .map_err(|e: XlsxError| e.to_string())?;
 434:        let approval_fmt_ref = match row.approval_label.as_str() {
 435:            "approved" => approved_fmt,
 436:            "rejected" => rejected_fmt,
 437:            "compat_shortlist" | "compat_context" => compat_fmt,
 438:            _ => review_fmt,
 439:        };
 440:        sheet
 441:            .write_string_with_format(
 442:                r,
 443:                COL_APPROVAL,
 444:                display_approval_label(&row.approval_label),
 445:                approval_fmt_ref,
 446:            )
 447:            .map_err(|e: XlsxError| e.to_string())?;
 448:        sheet
 449:            .write_string_with_format(
 450:                r,
 451:                COL_HARD_GATE,
 452:                if row.hard_gate_passed == Some(true) {
 453:                    "pass"
 454:                } else if row.hard_gate_passed == Some(false) {
 455:                    "fail"
 456:                } else {
 457:                    "compat"
 458:                },
 459:                text_fmt,
 460:            )
 461:            .map_err(|e: XlsxError| e.to_string())?;
 462:        sheet
 463:            .write_string_with_format(r, COL_PROFILE, &row.profile_kind, text_fmt)

 467:            .map_err(|e: XlsxError| e.to_string())?;

 470:            .map_err(|e: XlsxError| e.to_string())?;

 475:            sheet,
 476:            r,

 484:        sheet
 485:            .write_number_with_format(r, COL_SHARPE, row.sharpe, num_fmt)
 486:            .map_err(|e: XlsxError| e.to_string())?;
 487:        sheet
 488:            .write_number_with_format(r, COL_MAX_DD, row.max_dd, num_fmt)
