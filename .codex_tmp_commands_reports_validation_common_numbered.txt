 498:    if !fast_ma.is_empty() || !slow_ma.is_empty() {
 499:        parts.push(format!(

 501:            if fast_ma.is_empty() { "?" } else { fast_ma },
 502:            if fast_len == 0 {
 503:                "?".to_string()
 504:            } else {
 505:                fast_len.to_string()
 506:            },
 507:            if slow_ma.is_empty() { "?" } else { slow_ma },
 508:            if slow_len == 0 {
 509:                "?".to_string()
 510:            } else {
 511:                slow_len.to_string()
 512:            }
 513:        ));
 514:    }
 515:
 516:    let mut profile_parts: Vec<String> = Vec::new();
 517:    for key in ACTIVE_PROFILE_KEYS {
 518:        if let Some(obj) = params.get(key).and_then(Value::as_object) {
 519:            let rendered = summarize_object(obj, 3);
 520:            if rendered.is_empty() {

 522:            } else {
 523:                profile_parts.push(format!("{}({rendered})", key.to_ascii_uppercase()));
 524:            }
 525:        }
 526:    }

 528:        if params.get(key).and_then(Value::as_object).is_some() {
 529:            profile_parts.push(key.to_ascii_uppercase());
 530:        }
 531:    }

1214:fn plateau_stats_for_row(
1215:    row: &ExportCandidateRow,
1216:    surfaces: &std::collections::BTreeMap<PlateauGroupKey, PlateauGroupSurface>,
1217:    source_label: &str,
1218:) -> Option<PlateauStats> {
1219:    let params_value: Value = serde_json::from_str(&row.params_json).ok()?;
1220:    let params = params_value.as_object()?;
1221:    let (group, fast_len, slow_len) = plateau_group_from_params(params, Some(&row.profile_kind))?;
1222:    let surface = surfaces.get(&group)?;
1223:    if surface.points.len() < 3 {
1224:        return None;
1225:    }
1226:
1227:    let fast_idx = surface
1228:        .fast_periods
1229:        .iter()
1230:        .position(|value| *value == fast_len)?;
1231:    let slow_idx = surface
1232:        .slow_periods
1233:        .iter()
1234:        .position(|value| *value == slow_len)?;
1235:    let fast_lo = fast_idx.saturating_sub(1);
1236:    let fast_hi = (fast_idx + 1).min(surface.fast_periods.len().saturating_sub(1));
1237:    let slow_lo = slow_idx.saturating_sub(1);
1238:    let slow_hi = (slow_idx + 1).min(surface.slow_periods.len().saturating_sub(1));
1239:    let fast_window = &surface.fast_periods[fast_lo..=fast_hi];
1240:    let slow_window = &surface.slow_periods[slow_lo..=slow_hi];
1241:
1242:    let neighbors: Vec<&PlateauSurfacePoint> = surface
1243:        .points
1244:        .iter()
1245:        .filter(|point| {
1246:            fast_window.contains(&point.fast_len) && slow_window.contains(&point.slow_len)
1247:        })
1248:        .collect();
1249:    if neighbors.len() < 2 {
1250:        return None;
1251:    }
1252:
1253:    let mut sharpe_values: Vec<f64> = neighbors.iter().map(|point| point.sharpe).collect();
1254:    let sharpe_median = median_f64(&mut sharpe_values)?;
1255:    let max_dd_worst = neighbors
1256:        .iter()
1257:        .map(|point| point.max_dd)
1258:        .fold(f64::NEG_INFINITY, f64::max);
1259:    if !max_dd_worst.is_finite() {
1260:        return None;
1261:    }
1262:
1263:    let support_strength = (neighbors.len().min(6) as f64) / 6.0;
1264:    let sharpe_stability = if row.sharpe.abs() > 1e-9 {
1265:        (sharpe_median / row.sharpe).clamp(0.0, 1.0)
1266:    } else if sharpe_median > 0.0 {
1267:        1.0
1268:    } else {
1269:        0.0
1270:    };
1271:    let dd_stability = if max_dd_worst.abs() <= 1e-12 {
1272:        1.0
1273:    } else {
1274:        (row.max_dd / max_dd_worst).clamp(0.0, 1.0)
1275:    };
1276:    let score = 0.45 * support_strength + 0.35 * sharpe_stability + 0.20 * dd_stability;
1277:
1278:    Some(PlateauStats {
1279:        score,
1280:        label: label_plateau(score, neighbors.len()),
1281:        source: source_label.to_string(),
1282:        neighbor_count: neighbors.len(),
1283:        sharpe_median,
1284:        max_dd_worst,
1285:    })
1286:}
